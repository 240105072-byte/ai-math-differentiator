const SYSTEM_PROMPTS = {
    level1: `You are an encouraging math teacher. Generate a basic Quadratic Equations problem for a beginner student.
    Requirements:
    1. Problem Statement: Use simple, small integer coefficients where roots are easy to find (e.g., x^2 - 5x + 6 = 0).
    2. Guided Hints: Provide step-by-step breakdown (e.g., identifying a, b, c, factoring or using the quadratic formula).
    3. Complete Solution: Detailed, step-by-step resolution written in simple English.`,

    level2: `You are a high school math teacher. Generate a standard grade-level Quadratic Equations problem.
    Requirements:
    1. Problem Statement: A standard curriculum problem (can include negative roots or non-zero right side, e.g., 2x^2 + 3x - 5 = 0).
    2. Formula Hints: Mention key formulas (Discriminant D = b^2 - 4ac) without giving away the full answer immediately.
    3. Complete Solution: Clear and accurate step-by-step solution.`,

    level3: `You are an advanced math instructor. Generate a complex, real-world application challenge involving Quadratic Equations.
    Requirements:
    1. Real-World Scenario: Frame the problem around physics (projectile motion), economics (profit optimization), or geometry (area optimization).
    2. Scaffolding & Questions: Provide guiding multi-step logic questions instead of direct hints.
    3. Complete Analytical Solution: Detailed mathematical derivation and explanation of the practical context.`
};

document.getElementById('generateBtn').addEventListener('click', async () => {
    const apiKey = document.getElementById('apiKey').value.trim();
    const selectedLevel = document.querySelector('input[name="level"]:checked').value;
    const outputDiv = document.getElementById('output');
    const contentDiv = document.getElementById('content');
    const loadingDiv = document.getElementById('loading');

    if (!apiKey) {
        alert("Please enter your OpenAI API Key!");
        return;
    }

    outputDiv.classList.remove('hidden');
    loadingDiv.classList.remove('hidden');
    contentDiv.innerHTML = '';

    try {
        const response = await fetch('https://api.openai.com/v1/chat/completions', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'Authorization': `Bearer ${apiKey}`
            },
            body: JSON.stringify({
                model: 'gpt-4o-mini',
                messages: [
                    { role: 'system', content: SYSTEM_PROMPTS[selectedLevel] },
                    { role: 'user', content: 'Generate a new unique quadratic equation task.' }
                ],
                temperature: 0.7
            })
        });

        const data = await response.json();

        if (data.error) {
            contentDiv.innerHTML = `<span style="color:red;">Error: ${data.error.message}</span>`;
        } else {
            contentDiv.innerText = data.choices[0].message.content;
        }
    } catch (err) {
        contentDiv.innerHTML = `<span style="color:red;">Failed to connect to API: ${err.message}</span>`;
    } finally {
        loadingDiv.classList.add('hidden');
    }
});
